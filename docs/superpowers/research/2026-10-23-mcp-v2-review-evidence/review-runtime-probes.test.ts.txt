import { describe, it, expect, vi } from 'vitest';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { AutomationClient } from '../../cli/src/client';
import { startStandaloneServer } from '../../../server/automation/standalone';
import { LocalProjectRepository } from '../../../server/automation/project-repository';
import { CommitService } from '../../../server/automation/commit-service';
import { saveProject, getProject } from '../../../server/storage';
import { LocalBackend } from '@/backend/local-backend';
import { getBaseRevision, resetSaveRevisionForTest } from '@/backend/save-revision';
import { publishDeployOperation } from '../src/operations/publish';
import { createPlan } from '../src/plan';
import { createSnapshot, createOperationContext } from '../src/session';
import { validatePlan } from '../src/validate';
import { checkFile } from '@/code/oracle/check-file';
import { catalogPlaceOperation } from '../src/operations/catalog';

const scope = { projectId: 'review-fixture', branchId: 'main' };
const page = `'use client'; export default function Page(){return <div data-id="root" style={{width:'100%',height:'auto'}}/>;}`;
function fixture() {
  const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'revyme-review-'));
  fs.mkdirSync(path.join(dir, 'projects'));
  const file = path.join(dir, 'projects', scope.projectId + '.json');
  fs.writeFileSync(file, JSON.stringify({ id:scope.projectId, name:'Review', revision:1,
    data:{format:'revyme-v1',files:{'app/page.client.tsx':page}}, receipts:{} }));
  return { dir, file, repo:new LocalProjectRepository(dir) };
}
function planFor(content: string, operationId='review-op') {
  return createPlan({ scope, baseRevision:1, operationId,
    changes:[{path:'app/page.client.tsx',beforeHash:null,content}] });
}

describe('Independent review characterization probes', () => {
  it('real HTTP snapshot is incompatible with AutomationClient', async () => {
    const f=fixture(), server=await startStandaloneServer(0,f.dir);
    try {
      const address=server.server.address() as {port:number};
      const client=new AutomationClient({apiUrl:`http://127.0.0.1:${address.port}`});
      const res=await client.getSnapshot(scope);
      expect(res.ok).toBe(false);
      console.log('REVIEW_SNAPSHOT',JSON.stringify(res));
      client.getSnapshot=async()=>({ok:true,data:(await f.repo.getSnapshot(scope))!,meta:{revision:1,commitStatus:'not_committed'}});
      const mutation=await client.execute(scope,'pages.create',{slug:'about',title:'About'});
      expect(mutation.ok).toBe(false);
      expect(JSON.parse(fs.readFileSync(f.file,'utf8')).revision).toBe(1);
      console.log('REVIEW_MUTATION',JSON.stringify(mutation));
    } finally { await server.close(); fs.rmSync(f.dir,{recursive:true,force:true}); }
  });
  it('same committed plan is rejected by the commit service on retry', async () => {
    const f=fixture();
    try {
      const service=new CommitService(f.repo),plan=planFor(page.replace('/>','>Saved</div>'));
      expect((await service.commit(plan)).ok).toBe(true);
      const retry=await service.commit(plan);
      expect(retry.ok).toBe(false);
      expect(retry.error?.code).toBe('REVISION_CONFLICT');
      expect(await service.status(scope.projectId,plan.operationId)).not.toBeNull();
      console.log('REVIEW_RETRY',JSON.stringify(retry));
    } finally { fs.rmSync(f.dir,{recursive:true,force:true}); }
  });
  it('failed disk replace leaves a successful receipt cached and skips retry write', async () => {
    const f=fixture(); const plan=planFor(page.replace('/>','>Saved</div>'));
    const rename=vi.spyOn(fs,'renameSync').mockImplementationOnce(()=>{throw new Error('simulated disk failure');});
    try {
      await expect(f.repo.applyPlan(plan)).rejects.toThrow('simulated disk failure');
      rename.mockRestore();
      const retry=await f.repo.applyPlan(plan);
      expect(retry.revision).toBe(2);
      expect(JSON.parse(fs.readFileSync(f.file,'utf8')).revision).toBe(1);
      console.log('REVIEW_FALSE_RECEIPT',JSON.stringify({receiptRevision:retry.revision,diskRevision:1}));
    } finally {rename.mockRestore();fs.rmSync(f.dir,{recursive:true,force:true});}
  });
  it('an ordinary editor save deletes persisted operation receipts', async () => {
    const f=fixture();
    try {
      const plan=planFor(page.replace('/>','>Saved</div>'));
      await f.repo.applyPlan(plan);
      const persisted=JSON.parse(fs.readFileSync(f.file,'utf8'));
      saveProject(scope.projectId,persisted.data,'Review',f.dir,undefined,undefined,2);
      expect(await new LocalProjectRepository(f.dir).getReceipt(scope.projectId,plan.operationId)).toBeNull();
      console.log('REVIEW_RECEIPTS_DROPPED',true);
    } finally {fs.rmSync(f.dir,{recursive:true,force:true});}
  });
  it('new validator accepts a source rejected by the actual Oracle', () => {
    const invalid=`'use client'; export default function Page(){return <div data-id="root" style={{width:'100%',height:'auto',padding:'1.5rem'}}/>;}`;
    const snapshot=createSnapshot(scope,1,{'app/page.client.tsx':page});
    expect(validatePlan(snapshot,planFor(invalid)).ok).toBe(true);
    const issues=checkFile(invalid,{kind:'page',path:'app/page.client.tsx'});
    expect(issues.some(i=>/spacing|pixels|rem/i.test(i.message))).toBe(true);
    console.log('REVIEW_ORACLE_BYPASS',JSON.stringify(issues.map(i=>({code:i.code,message:i.message}))));
  });
  it('catalog places a component import without installing that component', async () => {
    const snapshot=createSnapshot(scope,1,{'app/page.client.tsx':page});
    const placed=await catalogPlaceOperation.execute(snapshot,{targetFile:'app/page.client.tsx',parentId:'root',itemId:'cs-marquee'},createOperationContext());
    expect(placed.changes).toHaveLength(1);
    expect(placed.changes[0].content).toContain('@/components/Marquee');
    expect(snapshot.files.has('components/Marquee.tsx')).toBe(false);
    const plan=createPlan({scope,baseRevision:1,operationId:'review-built-in',changes:placed.changes});
    expect(validatePlan(snapshot,plan).ok).toBe(true);
    console.log('REVIEW_MISSING_BUILTIN',JSON.stringify({changedFiles:placed.changes.map(c=>c.path),validator:validatePlan(snapshot,plan)}));
  });
  it('real editor/backend serialization allows the first stale save to overwrite an MCP commit', async () => {
    const f=fixture();
    resetSaveRevisionForTest();
    localStorage.clear();
    vi.stubGlobal('fetch',async (_url: unknown, init?: RequestInit)=>{
      if (init?.method === 'PUT') {
        const body=JSON.parse(init.body as string);
        const saved=saveProject(scope.projectId,body.data,body.name,f.dir,undefined,undefined,body.baseRevision);
        return new Response(JSON.stringify({success:true,project:saved}));
      }
      return new Response(JSON.stringify(getProject(scope.projectId,f.dir)));
    });
    try {
      const backend=new LocalBackend();
      const loaded=(await backend.loadProject(scope.projectId))!;
      expect(getBaseRevision()).toBeNull();
      await f.repo.applyPlan(planFor(page.replace('/>','>Agent change</div>')));
      await backend.saveProject(scope.projectId,loaded);
      const disk=JSON.parse(fs.readFileSync(f.file,'utf8'));
      expect(disk.data.files['app/page.client.tsx']).toBe(page);
      console.log('REVIEW_GUI_CLOBBER',JSON.stringify({baseAfterLoad:null,diskRevision:disk.revision,agentChangeLost:true}));
    } finally {vi.unstubAllGlobals();resetSaveRevisionForTest();localStorage.clear();fs.rmSync(f.dir,{recursive:true,force:true});}
  });
  it('editor reload confuses local recovery timestamp with server CAS revision', async () => {
    const f=fixture();
    resetSaveRevisionForTest();localStorage.clear();
    vi.stubGlobal('fetch',async (_url: unknown, init?: RequestInit)=>{
      if (init?.method === 'PUT') {
        const body=JSON.parse(init.body as string);
        try {
          const saved=saveProject(scope.projectId,body.data,body.name,f.dir,undefined,undefined,body.baseRevision);
          return new Response(JSON.stringify({success:true,project:saved}));
        } catch(e:any) {return new Response(JSON.stringify({error:e.message}),{status:e.status ?? 500});}
      }
      return new Response(JSON.stringify(getProject(scope.projectId,f.dir)));
    });
    try {
      const backend=new LocalBackend();
      const loaded=(await backend.loadProject(scope.projectId))!;
      await backend.saveProject(scope.projectId,loaded);
      resetSaveRevisionForTest();localStorage.clear();
      const reloaded=(await backend.loadProject(scope.projectId))!;
      const base=getBaseRevision();
      expect(base).toBeGreaterThan(1_000_000);
      await expect(backend.saveProject(scope.projectId,reloaded)).rejects.toThrow('REVISION_CONFLICT');
      console.log('REVIEW_GUI_FALSE_CONFLICT',JSON.stringify({loadedBase:base,diskRevision:2}));
    } finally {vi.unstubAllGlobals();resetSaveRevisionForTest();localStorage.clear();fs.rmSync(f.dir,{recursive:true,force:true});}
  });
  it('deploy reports a published URL without any hosting adapter or network call', async () => {
    const snapshot=createSnapshot(scope,1,{'app/page.client.tsx':page});
    const fetchSpy=vi.fn(()=>{throw new Error('unexpected network call');});
    vi.stubGlobal('fetch',fetchSpy);
    try {
      const result=await publishDeployOperation.execute(snapshot,{target:'production',force:true},createOperationContext());
      expect(result.status).toBe('deployed');
      expect(result.success).toBe(true);
      expect(fetchSpy).not.toHaveBeenCalled();
      console.log('REVIEW_FAKE_DEPLOY',JSON.stringify({status:result.status,url:result.url,networkCalls:fetchSpy.mock.calls.length}));
    } finally {vi.unstubAllGlobals();}
  });
});
